\usepackage{booktabs} % pandoc emits \toprule/\midrule/\bottomrule for pipe tables

% Classic paragraphs: first-line indent, no gap. Vertical space is then
% reserved for environments, which makes their boundaries visible.
\setlength{\parindent}{1.5em}
\setlength{\parskip}{0pt plus 1pt}

% Space above and below theorem-like environments (amsthm hook).
\makeatletter
\def\thm@space@setup{%
  \thm@preskip=0.9\baselineskip plus 0.2\baselineskip minus 0.1\baselineskip
  \thm@postskip=\thm@preskip}
\makeatother

% Upright environments end with a lozenge, as proofs end with a box.
\newcommand{\EndMarkEnv}[1]{%
  \AtBeginEnvironment{#1}{\pushQED{\qed}\renewcommand{\qedsymbol}{$\lozenge$}}%
  \AtEndEnvironment{#1}{\popQED}}
\EndMarkEnv{definition}
\EndMarkEnv{example}
\EndMarkEnv{remark}
